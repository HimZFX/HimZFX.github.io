{{- /* 资料预览页的语言配置：构建时生成 /viewer/site.js。 预览页 static/viewer/ 是纯静态页面，读不到 Hugo 的配置，所以由这里把「默认语言是哪个、每种语言的网址前缀」 写成一个小脚本（例如 window.SITE_LANGUAGES = {"defaultLang":"en","languages":[{"lang":"en","prefix":""},{"lang":"zh","prefix":"zh"}]}）。 static/viewer/lang.js 用它决定：链接里没写 lang 参数时用哪种界面语言，左上角「主页」指向 / 还是 /zh/。 内容只来自 config/_default/ 的语言配置（hugo.toml 的 defaultContentLanguage、languages.toml），每个首页渲染时生成， 各语言写出的内容相同。不在页面里输出任何东西。 */ -}} {{- $langs := slice -}} {{- range site.Sites -}} {{- $langs = $langs | append (dict "lang" .Language.Lang "prefix" (strings.TrimPrefix "/" .LanguagePrefix)) -}} {{- end -}} {{- $data := dict "defaultLang" site.Sites.Default.Language.Lang "languages" $langs -}} {{- $js := printf "/* Generated by Hugo from config/_default/ (layouts/partials/features/viewer-site.html). Do not edit. */\nwindow.SITE_LANGUAGES = %s;\n" ($data | jsonify) -}} {{- (resources.FromString "viewer/site.js" $js).Publish -}}
